```c
// 初始化:
SX1276Write( REG_LR_OPMODE, RFLR_OPMODE_SLEEP );          // 进入睡眠模式
SX1276Write( REG_LR_OPMODE, 0x80|RFLR_OPMODE_SLEEP );     // 进入lora 模式 
SX1276Write( REG_LR_OPMODE, 0x80|RFLR_OPMODE_STANDBY );	  // lora待机模式

SX1276Write( REG_LR_OPMODE, 0x80|RFLR_OPMODE_STANDBY ); //待机模式
SX1276Write( REG_LR_FRFMSB, (uint8_t)(freq_reg>>16));//Carrier Freq 470M  Frf = Fstep x Frf(23;0)
SX1276Write( REG_LR_FRFMID, (uint8_t)(freq_reg>>8) );
SX1276Write( REG_LR_FRFLSB, (uint8_t)(freq_reg) );

SX1276Write( REG_LR_OPMODE, 0x80|RFLR_OPMODE_STANDBY ); //待机模式
SX1276Write( REG_LR_PACONFIG, 0xf0|GSX127X_LoRaConfig.PowerCfig);//PA_BOOST 引脚。最大功率为+20dBm

SX1276Write( REG_LR_PADAC, 0x80|RFLR_PADAC_20DBM_ON );  //PA_BOOST 上为+20dBm //SX1276Write( REG_LR_PADAC, 0x80|RFLR_PADAC_20DBM_OFF );
SX1276Write( REG_LR_PARAMP,RFLR_PARAMP_0100_US);  //电流斜率

SX1276Write( REG_LR_OCP,0x20|RFLR_OCP_TRIM_240_MA);//电流过载保护 Over Current Protection

SX1276Write( REG_LR_MODEMCONFIG1,BandWidth|CodingRate|ExplicitHeaderOn);//BW、CR、Header有无，初始化  

SX1276Write( REG_LR_MODEMCONFIG2,SpreadingFactor|CRCON); //SF、CRC初始化   

SX1276Write( REG_LR_MODEMCONFIG3,LOWDATARATEOPTIMIZE?|RFLR_MODEMCONFIG3_AGCAUTO_ON);//低速率优化功能是否开启、AutoAGC默认开启   
```

SX1276WriteRxTx(FALSE);

SX1276Write( REG_LR_FIFORXBASEADDR,0x00);    	
SX1276Write(REG_LR_IRQFLAGS,0xff);                          
SX1276Write( REG_LR_OPMODE, 0x80|RFLR_OPMODE_RECEIVER );
